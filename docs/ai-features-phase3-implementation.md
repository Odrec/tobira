```typescript
import React from "react";
import { useFragment, graphql } from "react-relay/hooks";
import { useTranslation } from "react-i18next";
import { Card } from "@opencast/appkit";
import { LuSparkles } from "react-icons/lu";

import { AiSummary$key } from "./__generated__/AiSummary.graphql";

const fragment = graphql`
  fragment AiSummary on AiSummary {
    summary
    model
    createdAt
    processingTimeMs
  }
`;

type Props = {
  fragmentRef: AiSummary$key;
};

export const AiSummary: React.FC<Props> = ({ fragmentRef }) => {
  const { t } = useTranslation();
  const data = useFragment(fragment, fragmentRef);

  if (!data) {
    return null;
  }

  return (
    <Card css={{
      padding: "1.5rem",
      marginTop: "1.5rem",
      backgroundColor: "var(--accent2)",
    }}>
      <div css={{
        display: "flex",
        alignItems: "center",
        marginBottom: "1rem",
        gap: "0.5rem",
      }}>
        <LuSparkles size={24} />
        <h3 css={{ margin: 0 }}>
          {t("video.ai-summary.title", "AI-Generated Summary")}
        </h3>
      </div>
      
      <div css={{
        lineHeight: 1.6,
        whiteSpace: "pre-wrap",
      }}>
        {data.summary}
      </div>
      
      <div css={{
        marginTop: "1rem",
        fontSize: "0.85rem",

        fontStyle: "italic",

      </div>
    </Card>
  );
};
```

### Step 3: Create AI Quiz Component

**File:** `frontend/src/ui/AiQuiz.tsx` (NEW)

```typescript
import React, { useState } from "react";
import { useFragment, graphql } from "react-relay/hooks";
import { useTranslation } from "react-i18next";
import { Card, Button } from "@opencast/appkit";
import { LuCheckCircle, LuXCircle, LuHelpCircle } from "react-icons/lu";

import { AiQuiz$key } from "./__generated__/AiQuiz.graphql";
import { QuizQuestion$data } from "./__generated__/AiQuiz.graphql";

const fragment = graphql`
  fragment AiQuiz on AiQuiz {
    questions {
      question
      questionType
      options
      correctAnswer
      explanation
      difficulty
      timestamp
    }
    model
    createdAt
  }
`;

type Props = {
  fragmentRef: AiQuiz$key;
  onSeekToTimestamp?: (seconds: number) => void;
};

export const AiQuiz: React.FC<Props> = ({ fragmentRef, onSeekToTimestamp }) => {
  const { t } = useTranslation();
  const data = useFragment(fragment, fragmentRef);
  const [currentQuestion, setCurrentQuestion] = useState(0);
  const [selectedAnswer, setSelectedAnswer] = useState<string | null>(null);
  const [showExplanation, setShowExplanation] = useState(false);
  const [score, setScore] = useState(0);
  const [answeredQuestions, setAnsweredQuestions] = useState<Set<number>>(new Set());

  if (!data || !data.questions || data.questions.length === 0) {
    return null;
  }

  const question = data.questions[currentQuestion];
  const isAnswered = answeredQuestions.has(currentQuestion);
  const isCorrect = selectedAnswer === question.correctAnswer;

  const handleAnswer = (answer: string) => {
    if (isAnswered) return;
    
    setSelectedAnswer(answer);
    setShowExplanation(true);
    
    if (answer === question.correctAnswer) {
      setScore(score + 1);
    }
    
    setAnsweredQuestions(new Set([...answeredQuestions, currentQuestion]));
  };

  const handleNext = () => {
    if (currentQuestion < data.questions.length - 1) {
      setCurrentQuestion(currentQuestion + 1);
      setSelectedAnswer(null);
      setShowExplanation(false);
    }
  };

  const handlePrevious = () => {
    if (currentQuestion > 0) {
      setCurrentQuestion(currentQuestion - 1);
      setSelectedAnswer(null);
      setShowExplanation(false);
    }
  };

  const handleSeek = () => {
    if (question.timestamp && onSeekToTimestamp) {
      onSeekToTimestamp(question.timestamp);
    }
  };

  return (
    <Card css={{
      padding: "1.5rem",
      marginTop: "1.5rem",
    }}>
      <div css={{
        display: "flex",
        justifyContent: "space-between",
        alignItems: "center",
        marginBottom: "1.5rem",
      }}>
        <h3 css={{ margin: 0 }}>
          <LuHelpCircle style={{ verticalAlign: "middle", marginRight: "0.5rem" }} />
          {t("video.ai-quiz.title", "Interactive Quiz")}
        </h3>
        <div css={{ fontSize: "0.9rem", color: "var(--grey40)" }}>
          {t("video.ai-quiz.score", "Score:")} {score}/{data.questions.length}
        </div>
      </div>

      <div css={{
        marginBottom: "1rem",
        fontSize: "0.85rem",
        color: "var(--grey40)",
      }}>
        {t("video.ai-quiz.question-number", "Question {{current}} of {{total}}", {
          current: currentQuestion + 1,
          total: data.questions.length,
        })}
        {" · "}
        <span css={{ textTransform: "capitalize" }}>{question.difficulty}</span>
      </div>

      <div css={{
        fontSize: "1.1rem",
        fontWeight: 500,
        marginBottom: "1.5rem",
      }}>
        {question.question}
      </div>

      <div css={{ display: "flex", flexDirection: "column", gap: "0.75rem" }}>
        {question.questionType === "true_false" ? (
          <>
            {["True", "False"].map(option => (
              <Button
                key={option}
                onClick={() => handleAnswer(option)}
                disabled={isAnswered}
                css={{
                  justifyContent: "flex-start",
                  padding: "1rem",
                  backgroundColor: isAnswered && selectedAnswer === option
                    ? (isCorrect ? "var(--success-color)" : "var(--danger-color)")
                    : undefined,
                }}
              >
                {option}
                {isAnswered && selectedAnswer === option && (
                  isCorrect ? <LuCheckCircle style={{ marginLeft: "auto" }} /> 
                           : <LuXCircle style={{ marginLeft: "auto" }} />
                )}
              </Button>
            ))}
          </>
        ) : (
          <>
            {question.options?.map((option, idx) => (
              <Button
                key={idx}
                onClick={() => handleAnswer(option)}
                disabled={isAnswered}
                css={{
                  justifyContent: "flex-start",
                  padding: "1rem",
                  backgroundColor: isAnswered && selectedAnswer === option
                    ? (isCorrect ? "var(--success-color)" : "var(--danger-color)")
                    : undefined,
                }}
              >
                {option}
                {isAnswered && selectedAnswer === option && (
                  isCorrect ? <LuCheckCircle style={{ marginLeft: "auto" }} /> 
                           : <LuXCircle style={{ marginLeft: "auto" }} />
                )}
              </Button>
            ))}
          </>
        )}
      </div>

      {showExplanation && (
        <div css={{
          marginTop: "1.5rem",
          padding: "1rem",
          backgroundColor: "var(--accent2)",
          borderRadius: "4px",
        }}>
          <strong>{t("video.ai-quiz.explanation", "Explanation:")}</strong>
          <p css={{ marginTop: "0.5rem", marginBottom: 0 }}>
            {question.explanation}
          </p>
        </div>
      )}

      <div css={{
        display: "flex",
        gap: "0.75rem",
        marginTop: "1.5rem",
        justifyContent: "space-between",
      }}>
        <div css={{ display: "flex", gap: "0.75rem" }}>
          <Button
            onClick={handlePrevious}
            disabled={currentQuestion === 0}
          >
            {t("video.ai-quiz.previous", "Previous")}
          </Button>
          <Button
            onClick={handleNext}
            disabled={currentQuestion === data.questions.length - 1}
          >
            {t("video.ai-quiz.next", "Next")}
          </Button>
        </div>
        
        {question.timestamp && (
          <Button onClick={handleSeek}>
            {t("video.ai-quiz.seek-to-topic", "Jump to topic in video")}
          </Button>
        )}
      </div>
    </Card>
  );
};
```

## Benefits of This Approach

1. **Type Safety:** Relay generates TypeScript types from GraphQL
2. **Performance:** GraphQL fetches exactly what's needed
3. **Consistency:** Uses Tobira's existing patterns
4. **Maintainability:** Components are isolated and reusable
5. **User Experience:** Progressive enhancement - works without AI data
